import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

public class programa {

    private static final String URL = "jdbc:mysql://localhost:3306/zapateria";
    private static final String USER = "root";
    private static final String PASSWORD = "root";

    public static void main(String[] args) {
        try {
            // 1. Establecer la conexión dentro del try
            Connection conexion = DriverManager.getConnection(URL, USER, PASSWORD);
            System.out.println("¡Conexión exitosa al esquema zapateria!");

            // 2. Verificar los argumentos pasados por terminal
            if (args.length > 0) {
                if (args[0].equals("InsertarDatosProveedor")) {
                    if (args.length != 5) {
                        throw new IllegalArgumentException("Faltan o sobran argumentos");
                    }
                    InsertarDatosProveedor(conexion, Long.parseLong(args[1]), args[2], args[3], Integer.parseInt(args[4]));

                } else if (args[0].equals("EliminarProveedor")) {
                    if (args.length != 2) {
                        throw new IllegalArgumentException("Faltan o sobran argumentos");
                    }
                    EliminarProveedor(conexion, Long.parseLong(args[1]));

                } else if (args[0].equals("RetornarProveedores")) {
                    if (args.length != 1) {
                        throw new IllegalArgumentException("Sobran argumentos");
                    }
                    RetornarProveedores(conexion);

                } else {
                    System.out.println("Comando no soportado");
                }
            } else {
                System.out.println("No se enviaron comandos como argumentos. Cerrando el programa.");
            }

            // 3. Cerrar la conexión al finalizar
            conexion.close();

        } catch (SQLException e) {
            System.out.println("Error al conectar con la base de datos:");
            e.printStackTrace();
        }
    }

    // Insertar datos de proveedores
    public static void InsertarDatosProveedor(Connection conexion, long CUIT, String razon_social, String direccion, int tel_fijo) {
        String sql = "INSERT INTO proveedor (CUIT, razon_social, direccion, tel_fijo) VALUES (?, ?, ?, ?)";
        
        try {
            PreparedStatement stmt = conexion.prepareStatement(sql);
            stmt.setLong(1, CUIT);
            stmt.setString(2, razon_social);
            stmt.setString(3, direccion);
            stmt.setInt(4, tel_fijo);
            
            stmt.executeUpdate();
            System.out.println("Proveedor insertado correctamente.");
            stmt.close();
            
        } catch (SQLException e) {
            e.printStackTrace();
        }
    }

    // Eliminar un proveedor dado su CUIT
    public static void EliminarProveedor(Connection conexion, long CUIT) {
        String sql = "DELETE FROM proveedor WHERE CUIT = ?";
        try {
            PreparedStatement stmt = conexion.prepareStatement(sql);
            stmt.setLong(1, CUIT);
            
            stmt.executeUpdate();
            System.out.println("Proveedor eliminado correctamente.");
            stmt.close();
            
        } catch (SQLException e) {
            e.printStackTrace();
        }
    }

    // Listar todos los proveedores
    public static void RetornarProveedores(Connection conexion) {
        String sql = "SELECT * FROM proveedor";
        try {
            PreparedStatement stmt = conexion.prepareStatement(sql);
            ResultSet rs = stmt.executeQuery();
            int count = 1;
            while (rs.next()) {
                System.out.println("Proveedor " + count + ": " + rs.getLong("CUIT") + ", " + rs.getString("razon_social") + ", " + rs.getString("direccion") + ", " + rs.getInt("tel_fijo"));
                count++;
            }
            rs.close();
            stmt.close();
            
        } catch (SQLException e) {
            e.printStackTrace();
        }
    }
}
